rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() { return request.auth != null; }
    function myMember() {
      return /databases/$(database)/documents/members/$(request.auth.uid);
    }
    function myData() { return get(myMember()).data; }
    function role() { return myData().get('role', 'member'); }
    function isAdmin() {
      return signedIn() && exists(myMember()) && role() in ['admin', 'owner'];
    }
    function isOwner() {
      return signedIn() && exists(myMember()) && role() == 'owner';
    }
    function isActive() {
      return exists(myMember()) && myData().get('status', 'active') == 'active';
    }
    function canUse(uid) {
      return signedIn() && request.auth.uid == uid && isActive();
    }
    // 대상(u)이 '차단된 일반 가입자'인지
    function isBlockedUid(u) {
      return exists(/databases/$(database)/documents/members/$(u))
        && get(/databases/$(database)/documents/members/$(u)).data.get('role', 'member') == 'member'
        && get(/databases/$(database)/documents/members/$(u)).data.get('status', 'active') == 'blocked';
    }
    function isPrivate(u) {
      return exists(/databases/$(database)/documents/users/$(u))
        && get(/databases/$(database)/documents/users/$(u)).data.get('private', false) == true;
    }
    function maxUsers() {
      return exists(/databases/$(database)/documents/config/settings)
        ? get(/databases/$(database)/documents/config/settings).data.maxUsers : 20;
    }
    function oldCount() {
      return exists(/databases/$(database)/documents/config/stats)
        ? get(/databases/$(database)/documents/config/stats).data.count : 0;
    }

    // ---- 소유자 등록 (처음 한 번만 만들 수 있음) ----
    match /config/owner {
      allow read: if signedIn();
      allow create: if signedIn()
        && !exists(/databases/$(database)/documents/config/owner)
        && request.resource.data.keys().hasOnly(['uid','at'])
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.at == request.time;
        // 선착순 위험까지 막으려면 위 줄 끝의 ';' 를 지우고 아래 두 줄을 추가하세요 (이메일은 본인 것으로):
        // && request.auth.token.email == '내이메일@gmail.com'
        // && request.auth.token.email_verified == true;
      allow update, delete: if false;
    }

    // ---- 인원 제한 ----
    match /config/settings {
      allow read: if true;
      allow write: if isAdmin()
        && request.resource.data.keys().hasOnly(['maxUsers'])
        && request.resource.data.maxUsers is int
        && request.resource.data.maxUsers >= 1
        && request.resource.data.maxUsers <= 1000;
    }
    match /config/stats {
      allow read: if true;
      allow create, update: if signedIn()
        && request.resource.data.keys().hasOnly(['count'])
        && request.resource.data.count is int
        && request.resource.data.count >= 0
        && (isAdmin()
            || (request.resource.data.count == oldCount() + 1
                && !exists(/databases/$(database)/documents/members/$(request.auth.uid))
                && existsAfter(/databases/$(database)/documents/members/$(request.auth.uid)))
            || (request.resource.data.count == oldCount() - 1
                && exists(/databases/$(database)/documents/members/$(request.auth.uid))
                && !existsAfter(/databases/$(database)/documents/members/$(request.auth.uid))));
    }

    // ---- 가입 정보 (역할·상태 포함) ----
    match /members/{uid} {
      allow get: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow list: if isAdmin();

      allow create: if signedIn() && request.auth.uid == uid
        && !exists(/databases/$(database)/documents/blocked/$(uid))
        && request.resource.data.keys().hasOnly(['joinedAt','status'])
        && request.resource.data.status == 'restricted'
        && request.resource.data.joinedAt == request.time
        && getAfter(/databases/$(database)/documents/config/stats).data.count == oldCount() + 1
        && getAfter(/databases/$(database)/documents/config/stats).data.count <= maxUsers();

      allow create, update: if signedIn() && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['joinedAt','role','status'])
        && request.resource.data.role == 'owner'
        && request.resource.data.get('status', 'active') == 'active'
        && getAfter(/databases/$(database)/documents/config/owner).data.uid == request.auth.uid;

      allow create: if isOwner() && uid != request.auth.uid
        && request.resource.data.keys().hasOnly(['joinedAt','role'])
        && request.resource.data.role == 'admin'
        && request.resource.data.joinedAt == request.time;

      allow update: if isOwner() && uid != request.auth.uid
        && resource.data.get('role', 'member') != 'owner'
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['role','status'])
        && request.resource.data.get('role', 'member') in ['admin', 'member']
        && request.resource.data.get('status', 'active') in ['active', 'restricted', 'blocked'];

      allow update: if isAdmin() && uid != request.auth.uid
        && resource.data.get('role', 'member') == 'member'
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status'])
        && request.resource.data.status in ['active', 'restricted', 'blocked'];

      allow delete: if
        (isAdmin() && resource.data.get('role', 'member') == 'member')
        || (isOwner() && uid != request.auth.uid)
        || (signedIn() && request.auth.uid == uid
            && resource.data.get('role', 'member') == 'member'
            && resource.data.get('status', 'active') != 'blocked');
    }

    // ---- 재가입 차단 기록 (이름은 알아보기 위한 선택 값) ----
    match /blocked/{uid} {
      allow get: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow list: if isAdmin();
      allow create, update: if isAdmin()
        && request.resource.data.keys().hasOnly(['at','name'])
        && (!('name' in request.resource.data)
            || (request.resource.data.name is string
                && request.resource.data.name.size() >= 1
                && request.resource.data.name.size() <= 20));
      allow delete: if isAdmin();
    }

    // ---- 사용자 프로필 ----
    match /users/{uid} {
      allow get: if true;
      allow list: if resource.data.listed == true;

      allow create, update: if canUse(uid)
        && request.resource.data.keys().hasOnly(['name','listed','private'])
        && (!('name' in request.resource.data)
            || (request.resource.data.name is string
                && request.resource.data.name.size() >= 1
                && request.resource.data.name.size() <= 20))
        && (!('listed' in request.resource.data) || request.resource.data.listed is bool)
        && (!('private' in request.resource.data) || request.resource.data.private is bool)
        && (request.resource.data.get('listed', false) == false
            || 'name' in request.resource.data)
        && !(request.resource.data.get('private', false) == true
             && request.resource.data.get('listed', false) == true);

      // 제한 중인 가입자: 프로필이 아직 없을 때 이름을 한 번만 만들 수 있어요
      allow create: if signedIn() && request.auth.uid == uid
        && exists(myMember())
        && request.resource.data.keys().hasOnly(['name'])
        && request.resource.data.name is string
        && request.resource.data.name.size() >= 1
        && request.resource.data.name.size() <= 20;

      // 본인 삭제(탈퇴) 또는 소유자의 '차단된 계정' 완전 삭제
      allow delete: if canUse(uid) || (isOwner() && isBlockedUid(uid));
    }

    // ---- 개인 일정 (비공개면 본인만 읽기 가능) ----
    match /users/{uid}/months/{month} {
      // 소유자는 '차단된 계정'의 일정만 완전 삭제를 위해 읽을 수 있어요
      allow read: if (signedIn() && request.auth.uid == uid)
        || !isPrivate(uid)
        || (isOwner() && isBlockedUid(uid));
      allow create, update: if canUse(uid)
        && month.matches('(20[0-9]{2}|21[0-9]{2})-(0[1-9]|1[0-2])')
        && request.resource.data.keys().hasOnly(['events','expireAt'])
        && request.resource.data.events is map
        && request.resource.data.events.size() <= 31
        && (!('expireAt' in request.resource.data)
            || request.resource.data.expireAt is timestamp);
      allow delete: if canUse(uid) || (isOwner() && isBlockedUid(uid));
    }

    // ---- 출시 일정 (관리자만) ----
    match /base/{month} {
      allow read: if true;
      allow create, update: if isAdmin()
        && month.matches('(20[0-9]{2}|21[0-9]{2})-(0[1-9]|1[0-2])')
        && request.resource.data.keys().hasOnly(['events','expireAt'])
        && request.resource.data.events is map
        && (!('expireAt' in request.resource.data)
            || request.resource.data.expireAt is timestamp);
      allow delete: if isAdmin();
    }
  }
}
