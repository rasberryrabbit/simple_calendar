rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    match /users/{uid} {
      allow get: if true;
      allow list: if resource.data.listed == true;
      allow write: if request.auth != null
        && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['name','listed'])
        && request.resource.data.name is string
        && request.resource.data.name.size() >= 1
        && request.resource.data.name.size() <= 20
        && (!('listed' in request.resource.data)
            || request.resource.data.listed is bool);
    }

    match /users/{uid}/months/{month} {
      allow read: if true;
      allow create, update: if request.auth != null
        && request.auth.uid == uid
        && month.matches('[0-9]{4}-[0-9]{2}')
        && request.resource.data.keys().hasOnly(['events','expireAt'])
        && request.resource.data.events is map
        && request.resource.data.events.size() <= 31
        && (!('expireAt' in request.resource.data)
            || request.resource.data.expireAt is timestamp);
      allow delete: if request.auth != null && request.auth.uid == uid;
    }

    match /base/{month} {
      allow read: if true;
      allow create, update: if request.auth != null
        && request.auth.uid in ['관리자 UID']
        && month.matches('[0-9]{4}-[0-9]{2}')
        && request.resource.data.keys().hasOnly(['events','expireAt'])
        && request.resource.data.events is map
        && (!('expireAt' in request.resource.data)
            || request.resource.data.expireAt is timestamp);
      allow delete: if request.auth != null
        && request.auth.uid in ['관리자 UID'];
    }
  }
}
